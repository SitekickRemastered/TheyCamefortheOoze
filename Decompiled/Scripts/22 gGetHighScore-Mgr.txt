property pNetXtra, pHighScoreURL, pGetText, pHighScoreXMLtextMem, pHighScoreTextMem, pNumScorestoRetrieve, pXMLXtra
global gHighScoreXML, gNames1pGlobal, gScores1pGlobal, gLevels1pGlobal

on new me
  pNumScorestoRetrieve = 10
  pHighScoreURL = "http://www.ytv.com/common/inc/CDROMfunctions/GetHighScore.asp?campaignID=1&TopNumber=" & pNumScorestoRetrieve
  pHighScoreXMLtextMem = member("HighScoreXML")
  pHighScoreTextMem = member("HighScore")
  pHighScoreXMLtextMem.text = EMPTY
  pHighScoreTextMem.text = EMPTY
  return me
end

on Init me
  if not _player.netPresent then
    _player.alert("Sorry, the Network Support Xtras could not be found.")
  else
    pNetXtra = new xtra("netlingo")
    clearCache()
  end if
end

on Uninit me
  case pNetXtra = VOID of
    0:
      pNetXtra = VOID
  end case
  case pXMLXtra = VOID of
    0:
      pNetXtra = VOID
  end case
  pHighScoreXMLtextMem.text = EMPTY
  pHighScoreTextMem.text = EMPTY
end

on RetrieveHighScores me
  pHighScoreXMLtextMem.text = EMPTY
  url = pHighScoreURL
  gHighScoreXML = getNetText(url)
end

on DisplayHighScoreXML me, artext
  pHighScoreXMLtextMem.text = artext
  pNetXtra = VOID
end

on parseHighScoreXML me
  pXMLXtra = new xtra("xmlparser")
  lvar = pXMLXtra.parseString(pHighScoreXMLtextMem.text)
  lvar2 = pXMLXtra.makeList()
  lcount = lvar2[1][2][2].count - 1
  case lvar of
    VOID:
      lpasslist = []
      repeat with i = 1 to lcount
        lhighscoreID = pXMLXtra.child[1].child[1].child[i].attributeValue[1]
        ltemplist = pXMLXtra.child[1].child[1].child[i].makeSublist()
        tName = getaProp(getaProp(getaProp(ltemplist, "highscore"), "username"), "!CHARDATA")
        tScore = getaProp(getaProp(getaProp(ltemplist, "highscore"), "score"), "!CHARDATA")
        tLevel = getaProp(getaProp(getaProp(ltemplist, "highscore"), "level"), "!CHARDATA")
        if voidp(tName) = 1 then
          lname = "SiteKick"
        else
          lname = ltemplist[1][2][2]
        end if
        if voidp(tScore) = 1 then
          lscore = "No Data"
        else
          lscore = ltemplist[1][3][2]
        end if
        if voidp(tLevel) = 1 then
          lLevel = "No Data"
        else
          lLevel = ltemplist[1][4][2]
        end if
        ltemplist2 = [lhighscoreID, lname, lscore, lLevel]
        lpasslist.append(ltemplist2)
      end repeat
    otherwise:
      trace("parse error")
  end case
  pXMLXtra = VOID
  me.DisplayHighScore(lpasslist)
end

on DisplayHighScore me, arlist
  gNames1pGlobal = []
  gScores1pGlobal = []
  gLevels1pGlobal = []
  pHighScoreTextMem.text = EMPTY
  lcount = arlist.count
  repeat with lx = 1 to lcount
    lstring = EMPTY
    lstring = lstring & "Rank:" && arlist[lx][1] & RETURN
    lstring = lstring & "Name:" && arlist[lx][2] & RETURN
    lstring = lstring & "Score:" && arlist[lx][3] & RETURN
    lstring = lstring & "Level:" && arlist[lx][4] & RETURN & RETURN
    pHighScoreTextMem.text = pHighScoreTextMem.text & lstring
    gNames1pGlobal[lx] = arlist[lx][2]
    gScores1pGlobal[lx] = arlist[lx][3]
    gLevels1pGlobal[lx] = arlist[lx][4]
  end repeat
end

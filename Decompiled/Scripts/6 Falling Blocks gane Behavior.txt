on playSound aChannel, soundName
  cantFind = 0
  if aChannel = 0 then
    playedSound = 0
    a = 3
    repeat while playedSound = 0
      if sound(a).status = 0 then
        sound(a).playFile(_movie.path & "sound\" & soundName)
        playedSound = 1
      end if
      a = a + 1
      if a = 9 then
        a = 3
      end if
      cantFind = cantFind + 1
      if cantFind = 6 then
        exit
      end if
    end repeat
  else
    sound(aChannel).playFile(_movie.path & "sound\" & soundName)
  end if
end

on spinMeter groupFormed, theLength, player
  startPower = gSpinMeter[player]
  sprite(8).visible = 1
  sprite(9).visible = 1
  currentMs = _system.milliSeconds
  if currentMs > (pLastSpinTime[player] + pSpinDecay[player]) then
    if gSpinMeter[player] > 0 then
      gSpinMeter[player] = gSpinMeter[player] - 1
    end if
    pLastSpinTime[player] = currentMs
  end if
  if groupFormed = 1 then
    if pChain[player] > 0 then
      gLastSpin[player] = pChain[player] * (pStandardSpin + ((theLength - pGroupSize) * pBonusSpin))
      gLastSpin[player] = gAttackRingChain[player] * gLastSpin[player]
    else
      if pConsecutive[player] > 0 then
        gLastSpin[player] = gLastSpin[player] + pStandardSpin + ((theLength - pGroupSize) * pBonusSpin)
        gLastSpin[player] = gAttackRingConsec[player] * gLastSpin[player]
      else
        gLastSpin[player] = pStandardSpin + ((theLength - pGroupSize) * pBonusSpin)
        gLastSpin[player] = gAttackRingCombo[player] * gLastSpin[player]
      end if
    end if
    gLastSpin[player] = gLastSpin[player] * gWeightDiscMass[player]
    put "lastspin " & gLastSpin[player]
    gSpinMeter[player] = gSpinMeter[player] + gLastSpin[player]
  end if
  if groupFormed = 2 then
    gSpinMeter[player] = gSpinMeter[player] + (theLength * pADBonusSpin)
  end if
  if gTestMode = 1 then
    tempString = member("lastText" & player).text
    if groupFormed <> 0 then
      member("lastText" & player).text = tempString & "Last power: " & gSpinMeter[player] - startPower
    end if
  end if
  if player = 1 then
    spinSprite = 8
  else
    spinSprite = 9
  end if
  if gSpinMeter[player] < 100 then
    sprite(spinSprite).width = pSpinMax * gSpinMeter[player] / 100
  else
    sprite(spinSprite).width = pSpinMax
  end if
  if g2Play = 0 then
    sprite(8).loc = point(404, 517)
    sprite(8).height = 19
    if gSpinMeter[player] < 100 then
      sprite(8).locH = 245 + (pSpinMax * gSpinMeter[player] / 100) - (pSpinMax * gSpinMeter[player] / 200)
    else
      sprite(8).locH = 245 + (pSpinMax * 100 / 100) - (pSpinMax * 100 / 200)
    end if
  else
    if player = 1 then
      if gSpinMeter[player] < 100 then
        sprite(8).locH = 434 + (pSpinMax * gSpinMeter[player] / 100) - (pSpinMax * gSpinMeter[player] / 200)
      else
        sprite(8).locH = 434 + (pSpinMax * 100 / 100) - (pSpinMax * 100 / 200)
      end if
      sprite(8).locV = 518
      sprite(8).height = 19
    else
      if gSpinMeter[player] < 100 then
        sprite(9).locH = 46 + (pSpinMax * gSpinMeter[player] / 100) - (pSpinMax * gSpinMeter[player] / 200)
      else
        sprite(9).locH = 46 + (pSpinMax * 100 / 100) - (pSpinMax * 100 / 200)
      end if
      sprite(9).locV = 518
      sprite(9).height = 19
    end if
  end if
end

on dropNewPiece me, player
  if gTutorial = 0 then
    pPieceType[player] = random(pPieceList[player].count)
  else
    pPieceType[player] = 1
  end if
  gClustersDropped[player] = gClustersDropped[player] + 1
  discColourList = [:]
  pPieceSprites[player] = []
  numSpritesNeeded = pPieceList[player][pPieceType[player]][1].count
  repeat with a = 1 to 10
    sprite(360 + a).width = pDiscWidth + 4
    sprite(360 + a).height = pDiscHeight + 4
  end repeat
  if gBBmeter[player] > 6 then
    gBBmeter[player] = 6
  end if
  numAttackDiscs = (gBBmeter[player] + 1) / 2
  attackDiscPos = numSpritesNeeded - numAttackDiscs
  repeat while pPieceSprites[player].count < numSpritesNeeded
    if (gAttackDisc[player] > 0) and (pPieceSprites[player].count = attackDiscPos) then
      pAttackType[player] = gAttackDisc[player]
      gAttackDisc[player] = 0
      gBBmeter[player] = 0
      repeat with i = 1 to numAttackDiscs
        usedDisc = 1
        inTheList = 1
        repeat while (usedDisc <> 0) and (inTheList <> 0)
          ranAttackDisc = random(10) + 460
          usedDisc = pActiveDiscs[1].getPos(ranAttackDisc) + pActiveDiscs[2].getPos(ranAttackDisc)
          inTheList = pPieceSprites[1].getPos(ranAttackDisc) + pPieceSprites[2].getPos(ranAttackDisc)
        end repeat
        sprite(ranAttackDisc).member = "attackDisc" & pAttackType[player]
        add(pPieceSprites[player], ranAttackDisc)
        add(pActiveDiscs[player], ranAttackDisc)
        put "DERTY" & sprite(ranAttackDisc).width & sprite(ranAttackDisc).height
      end repeat
      next repeat
    end if
    ranPieceChoose = random(300) + pFirstSprite - 1
    ranColour = random(pNumColours)
    case ranColour of
      1:
        ranColourName = "red"
      2:
        ranColourName = "yellow"
      3:
        ranColourName = "blue"
      4:
        ranColourName = "green"
      5:
        ranColourName = "purple"

on drawPiece me, player
  repeat with i = 1 to pPieceSprites[player].count
    theX = pPieceList[player][pPieceType[player]][pPieceOrientation[player]][i][1] * pDiscWidth
    theY = pPieceList[player][pPieceType[player]][pPieceOrientation[player]][i][2] * pDiscHeight
    sprite(pPieceSprites[player][i]).loc = pPieceLoc[player] + point(theX, theY)
  end repeat
end

on movePiecesDown me, player, mType
  discList = []
  testagain = 1
  repeat while testagain = 1
    testagain = 0
    repeat with i in pActiveDiscs[player]
      if sprite(i).locV >= pFloor[player] then
        next repeat
      end if
      canMove = 1
      repeat with j in pActiveDiscs[player]
        if i = j then
          next repeat
        end if
        if sprite(j).locV >= pFloor[player] then
        end if
        if (sprite(i).locH = sprite(j).locH) and (sprite(j).locV = (sprite(i).locV + 32)) then
          canMove = 0
          exit repeat
        end if
      end repeat
      if canMove = 1 then
        if discList.getOne(i) = 0 then
          discList.add(i)
        end if
      end if
    end repeat
    if discList.count > 0 then
      repeat with disc in discList
        repeat with h = 1 to pDiscHeight
          sprite(disc).locV = sprite(disc).locV + 1
          if (offset("dam", sprite(disc).member.name) <> 0) or (sprite(disc).member = "bbdisc") then
            if (h mod 8) = 0 then
              updateStage()
            end if
            next repeat
          end if
          if (h mod 3) = 0 then
            updateStage()
          end if
        end repeat
      end repeat
      toggle = 1
      pDropDiscs[player] = 1
      if (toggle = 1) and (mType = 0) then
        return 0
      else
        discList = []
        testagain = 1
      end if
      next repeat
    end if
    testagain = 0
  end repeat
  pDropDiscs[player] = 0
  return 1
end

on checkForDisc disc, wantedPosition, sameDifferentColour, colour, player
  repeat with j in pActiveDiscs[player]
    if sameDifferentColour = 1 then
      if (j = disc) or (sprite(j).member.name <> colour) then
        next repeat
      end if
    else
      if (j = disc) or (colour.getPos(sprite(j).member.name) <> 0) then
        next repeat
      end if
    end if
    case wantedPosition of
      "right":
        if (sprite(j).locV = sprite(disc).locV) and (sprite(j).locH = (sprite(disc).locH + pDiscWidth)) then
          return j
        end if
      "below":
        if (sprite(j).locH = sprite(disc).locH) and (sprite(j).locV = (sprite(disc).locV + pDiscHeight)) then
          return j
        end if
      "left":
        if (sprite(j).locV = sprite(disc).locV) and (sprite(j).locH = (sprite(disc).locH - pDiscWidth)) then
          return j
        end if
      "above":
        if (sprite(j).locH = sprite(disc).locH) and (sprite(j).locV = (sprite(disc).locV - pDiscHeight)) then
          return j
        end if
    end case
  end repeat
  return 0
end

on levelUp me, player
  movePiecesDown(me, player, 1)
  bonusDiscList = []
  repeat with a = 500 to 650
    sprite(a).height = pDiscHeight
    sprite(a).width = pDiscWidth
    sprite(a).visible = 0
  end repeat
  a = 500
  theColumn = pFarLeft
  repeat while theColumn < (pFarRight + pDiscWidth)
    bonusDiscList.add(a)
    hitApiece = 0
    sprite(a).loc = pEntryLocation[player]
    sprite(a).locH = theColumn
    repeat while 1
      modTemp = pFloor[player] mod pDiscHeight
      if (sprite(a).locV mod pDiscHeight) = modTemp then
        repeat with s in pActiveDiscs[player]
          if (sprite(s).locH = sprite(a).locH) and (sprite(a).locV = (sprite(s).locV - pDiscHeight)) then
            hitApiece = 1
            exit repeat
          end if

    end repeat

    end if

  updateStage()
  repeat with i = 500 to a - 1

    showScore(me, player)
    bonusDiscList.deleteOne(i)
    updateStage()
    pauseIt(1)
  end repeat

  repeat with i = 500 to a - 1
    sprite(i).locV = -100
  end repeat

  sprite("levelup").locV = 220

on recursiveSearch theList, v, player
  i = v
  repeat with j in pActiveDiscs[player]
    if (i = j) or (theList.getPos(j) <> 0) or ((sprite(theList[1]).member <> sprite(j).member) and (j <= pLastSprite)) or (offset("dam", sprite(j).member.name) <> 0) then
      next repeat
    end if
    if (sprite(j).locV = sprite(i).locV) and (sprite(j).locH = (sprite(i).locH + pDiscWidth)) then
      theList.add(j)
      recursiveSearch(theList, j, player)
    end if
    if (sprite(j).locH = sprite(i).locH) and (sprite(j).locV = (sprite(i).locV + pDiscHeight)) then
      theList.add(j)
      recursiveSearch(theList, j, player)
    end if
    if (sprite(j).locV = sprite(i).locV) and (sprite(j).locH = (sprite(i).locH - pDiscWidth)) then
      theList.add(j)
      recursiveSearch(theList, j, player)
    end if
    if (sprite(j).locH = sprite(i).locH) and (sprite(j).locV = (sprite(i).locV - pDiscHeight)) then
      theList.add(j)
      recursiveSearch(theList, j, player)
    end if
  end repeat
  return theList
end

on exitFrame me
  if pState = 2 then
    pState = 0
    gState = 0
  end if
  if gCurrentMusic = 0 then
    musicPlayer(me, 0)
  end if
  gFrameCount = gFrameCount + 1
  if (gState <> 1) or ((gTutorial > 0) and (gState = 1)) then
    if (gTutorialNum = 6) and (pDamageDiscsH[2] > 0) and (pTempVar2 = 0) and (pPauseGame[1] = 0) then
      pTempVar2 = 1
      drop(me, 2)
      drawPiece(me, 2)
      put "gimme"
    end if
    if (gState <> 1) and (g2Play = 0) and (pPauseGame[1] = 0) then
      if (pDropDiscs[1] = 0) and (pKeysLocked[1] = 0) and (pPieceSprites[1].count > 0) then
        keyboardInput(me, 1)
      end if
    end if
    if (gState <> 1) and (g2Play = 1) then
      if (pDropDiscs[1] = 0) and (pKeysLocked[1] = 0) and (pPauseGame[1] = 0) and (pPieceSprites[1].count > 0) then
        keyboardInput(me, 1)
      end if
      if (pDropDiscs[2] = 0) and (pKeysLocked[2] = 0) and (pPauseGame[2] = 0) and (pPieceSprites[2].count > 0) then
        keyboardInput(me, 2)
      end if
    end if
    if pSmoothDrop = 0 then
      if g2Play = 0 then
        if (pDropDiscs[1] = 1) and (gState <> 1) then
          movePiecesDown(me, 1, 0)
        end if
        if the ticks > pNextMoveTime[1] then
          movePiece(me, 1)
          pNextMoveTime[1] = the ticks + pSpeed[1]
        else
          if (pDropDiscs[1] = 1) and (pPauseGame[1] = 0) then
            movePiecesDown(me, 1, 0)
          end if
        end if
      else
        if (pDropDiscs[1] = 1) and (gState <> 1) and (pPauseGame[1] = 0) then
          movePiecesDown(me, 1, 0)
        end if
        if (pDropDiscs[2] = 1) and (gState <> 1) and (pPauseGame[2] = 0) then
          movePiecesDown(me, 2, 0)
        end if
        if the ticks > pNextMoveTime[1] then
          movePiece(me, 1)
          pNextMoveTime[1] = the ticks + pSpeed[1]
        else
          if (pDropDiscs[1] = 1) and (pPauseGame[1] = 1) then
            movePiecesDown(me, 1, 0)
          end if
        end if
        if the ticks > pNextMoveTime[2] then
          movePiece(me, 2)
          pNextMoveTime[2] = the ticks + pSpeed[2]
        else
          if (pDropDiscs[2] = 1) and (pPauseGame[2] = 1) then
            movePiecesDown(me, 2, 0)
          end if
        end if
      end if
    else
      movePiece(me, 1)
      if g2Play = 1 then
        movePiece(me, 2)
      end if
    end if
  end if
  if (gFrameNum mod 10) = 0 then
    repeat with a = 441 to 460
      if sprite(a).locV < 0 then
        next repeat
      end if
    end repeat
  end if
  gFrameNum = gFrameNum + 1
  if gFrameNum = 2 then
    fps = integer(gFrameNum / ((the milliSeconds - gStartMilliSeconds) / 1000.0))
    member("FPS").text = fps & " fps"
    if (fps < pLowFPS) and (fps > 0) then
      pLowFPS = fps
      put "PLOW" & pLowFPS
      member("FPS2").text = string(pLowFPS) & " Min FPS"
    end if

    if gTestMode = 1 then
      sprite(55).loc = point(5, 1)
      sprite(56).loc = point(66, 1)
    end if
    gFrameNum = 0
  end if
  go(the frame)
end

on keyboardInput me, player
  if player = 1 then
    leftkey = 123
    rightkey = 124
    upKey = 126
    downKey = 125
    dropkey = SPACE
  else
    leftkey = 0
    rightkey = 2
    upKey = 13
    downKey = 1
    dropkey = TAB
  end if
  theMove = "none"
  currentMs = _system.milliSeconds
  if keyPressed(rightkey) then
    if (currentMs > (getaProp(pKeyboardmove[player], rightkey) + pKeySpeed)) and not hitOtherPiece(me, pDiscWidth, player) then
      pPieceLoc[player].locH = pPieceLoc[player].locH + pDiscWidth
      setaProp(pKeyboardmove[player], rightkey, _system.milliSeconds)
    end if
    pushAwayFromEdges(me, player)
    drawPiece(me, player)
    if checkForOverlap(me, player) = 1 then
      pPieceLoc[player].locH = pPieceLoc[player].locH - pDiscWidth
      pushAwayFromEdges(me, player)
      drawPiece(me, player)
    end if
    theMove = "right"
  else
    if keyPressed(leftkey) then
      if (currentMs > (getaProp(pKeyboardmove[player], leftkey) + pKeySpeed)) and not hitOtherPiece(me, -pDiscWidth, player) then
        pPieceLoc[player].locH = pPieceLoc[player].locH - pDiscWidth
        setaProp(pKeyboardmove[player], leftkey, _system.milliSeconds)
      end if
      pushAwayFromEdges(me, player)
      drawPiece(me, player)
      if checkForOverlap(me, player) = 1 then
        pPieceLoc[player].locH = pPieceLoc[player].locH + pDiscWidth
        pushAwayFromEdges(me, player)
        drawPiece(me, player)
      end if
      theMove = "left"
    end if
  end if
  if keyPressed(upKey) then
    if (currentMs > (getaProp(pKeyboardmove[player], upKey) + pKeySpeed)) and (pRotateCycle[player] = 0) then
      pPieceOrientation[player] = pPieceOrientation[player] + 1
      setaProp(pKeyboardmove[player], upKey, _system.milliSeconds)
      if pPieceOrientation[player] > pPieceList[player][pPieceType[player]].count then
        pPieceOrientation[player] = 1
      end if
      push = pushAwayFromEdges(me, player)
      drawPiece(me, player)
      if checkForOverlap(me, player) = 1 then
        pPieceOrientation[player] = pPieceOrientation[player] - 1
        if pPieceOrientation[player] < 1 then
          pPieceOrientation[player] = pPieceList[player][pPieceType[player]].count
        end if
        pushBackCluster(me, player, push)
        drawPiece(me, player)
      end if
    else
      if (currentMs > (getaProp(pKeyboardmove[player], upKey) + pKeySpeed)) and (pRotateCycle[player] = 1) then
        setaProp(pKeyboardmove[player], upKey, _system.milliSeconds)
        lastDisc = getLast(pPieceSprites[player])
        deleteAt(pPieceSprites[player], pPieceSprites[player].count)
        pPieceSprites[player].addAt(1, lastDisc)
        drawPiece(me, player)
      else
        if (currentMs > (getaProp(pKeyboardmove[player], upKey) + pKeySpeed)) and (pRotateCycle[player] = 2) then
          pPieceOrientation[player] = pPieceOrientation[player] - 1
          setaProp(pKeyboardmove[player], upKey, _system.milliSeconds)
          if pPieceOrientation[player] < 1 then
            pPieceOrientation[player] = pPieceList[player][pPieceType[player]].count
          end if
          push = pushAwayFromEdges(me, player)
          drawPiece(me, player)
          if checkForOverlap(me, player) = 1 then
            pPieceOrientation[player] = pPieceOrientation[player] + 1
            if pPieceOrientation[player] > pPieceList[player][pPieceType[player]].count then
              pPieceOrientation[player] = 1
            end if
            pushBackCluster(me, player, push)
            drawPiece(me, player)
          end if
        end if
      end if
    end if
    theMove = "up"
  end if
  if keyPressed(downKey) then
    if pSmoothDrop = 0 then
      if (currentMs > (getaProp(pKeyboardmove[player], downKey) + pKeySpeed)) and not (fallOnPiece(me, player) or hitBottom(me, player)) then
        pPieceLoc[player].locV = pPieceLoc[player].locV + pDiscHeight
        setaProp(pKeyboardmove[player], downKey, _system.milliSeconds)
      end if
      pushAwayFromEdges(me, player)
      drawPiece(me, player)
      if checkForOverlap(me, player) = 1 then
        pPieceLoc[player].locV = pPieceLoc[player].locV - pDiscHeight
        drawPiece(me, player)
      end if
    else
      modTemp = pFloor[player] mod pDiscHeight
      if ((sprite(pPieceSprites[player][1]).locV mod pDiscHeight) = modTemp) and (pClusterMoved[player] = 1) then
        pClusterMoved[player] = 0
        if fallOnPiece(me, player) or hitBottom(me, player) then
          exit
        end if
      end if
      pClusterMoved[player] = 1
      pPieceLoc[player].locV = pPieceLoc[player].locV + 1
      drawPiece(me, player)
      updateStage()
    end if
    theMove = "down"
  end if
  if keyPressed(dropkey) then
    put "SPACE"
    if currentMs > (getaProp(pKeyboardmove[player], dropkey) + pKeySpeed) then
      drop(me, player)
      drawPiece(me, player)
      setaProp(pKeyboardmove[player], dropkey, _system.milliSeconds)
      return "drop"
    end if
  end if
  return theMove
end

on keyDown2 me
  player = 1
  if gState <> 1 then
    currentMs = _system.milliSeconds
    case the keyCode of
      6:
        if currentMs > (getaProp(pKeyboardmove[player], 6) + pKeySpeed) then
          lastDisc = getLast(pPieceSprites[player])
          deleteAt(pPieceSprites[player], pPieceSprites[player].count)
          pPieceSprites[player].addAt(1, lastDisc)
        end if
        drawPiece(me, player)
      50:
        writeTempLog(me)
        writeLogFile(me)
        go("End")
        abort()
      124:
        if (currentMs > (getaProp(pKeyboardmove[player], 124) + pKeySpeed)) and not hitOtherPiece(me, pDiscWidth, player) then
          pPieceLoc[player].locH = pPieceLoc[player].locH + pDiscWidth
          setaProp(pKeyboardmove[player], 124, _system.milliSeconds)
        end if
        pushAwayFromEdges(me, player)
        drawPiece(me, player)
        if checkForOverlap(me, player) = 1 then
          pPieceLoc[player].locH = pPieceLoc[player].locH - pDiscWidth
          pushAwayFromEdges(me, player)
          drawPiece(me, player)
        end if
      123:
        if (currentMs > (getaProp(pKeyboardmove[player], 123) + pKeySpeed)) and not hitOtherPiece(me, -pDiscWidth, player) then
          pPieceLoc[player].locH = pPieceLoc[player].locH - pDiscWidth
          setaProp(pKeyboardmove[player], 123, _system.milliSeconds)
        end if
        pushAwayFromEdges(me, player)
        drawPiece(me, player)
        if checkForOverlap(me, player) = 1 then
          pPieceLoc[player].locH = pPieceLoc[player].locH + pDiscWidth
          pushAwayFromEdges(me, player)
          drawPiece(me, player)
        end if
      126:
        if (currentMs > (getaProp(pKeyboardmove[player], 126) + pKeySpeed)) and (pRotateCycle[player] = 0) then
          pPieceOrientation[player] = pPieceOrientation[player] + 1
          setaProp(pKeyboardmove[player], 126, _system.milliSeconds)
          if pPieceOrientation[player] > pPieceList[player][pPieceType[player]].count then
            pPieceOrientation[player] = 1
          end if
          pushAwayFromEdges(me, player)
          drawPiece(me, player)
          if checkForOverlap(me, player) = 1 then
            pPieceOrientation[player] = pPieceOrientation[player] - 1
            if pPieceOrientation[player] < 1 then
              pPieceOrientation[player] = pPieceList[player][pPieceType[player]].count
            end if
            pushAwayFromEdges(me, player)
            drawPiece(me, player)
          end if
        else
          lastDisc = getLast(pPieceSprites[player])
          deleteAt(pPieceSprites[player], pPieceSprites[player].count)
          pPieceSprites[player].addAt(1, lastDisc)
          drawPiece(me, player)
        end if
      125:
        if pSmoothDrop = 0 then
          if (currentMs > (getaProp(pKeyboardmove[player], 125) + pKeySpeed)) and not (fallOnPiece(me, player) or hitBottom(me, player)) then
            pPieceLoc[player].locV = pPieceLoc[player].locV + pDiscHeight
            setaProp(pKeyboardmove[player], 125, _system.milliSeconds)
          end if
          pushAwayFromEdges(me, player)
          drawPiece(me, player)
        else
          repeat with i = 1 to 20
            if not movePiece(me) then
              exit
            end if
            updateStage()
          end repeat
        end if
    end case
    case the key of
      SPACE:
        drop(me, player)
        pushAwayFromEdges(me, player)
        drawPiece(me, player)
    end case
  end if
end

on pushAwayFromEdges me, player
  repeat with v = 3 to pDiscsInCluster
    pastLeftEdge = 0
    pastRightEdge = 0
    pastTopEdge = 0
    pastBotEdge = 0
    repeat with i = 1 to pPieceSprites[player].count
      x = pPieceLoc[player].locH + (pPieceList[player][pPieceType[player]][pPieceOrientation[player]][i][1] * pDiscWidth)
      y = pPieceLoc[player].locV + (pPieceList[player][pPieceType[player]][pPieceOrientation[player]][i][2] * pDiscHeight)
      if x > pRightWall[player] then
        pastRightEdge = 1
      end if
      if x < pLeftWall[player] then
        pastLeftEdge = 1
      end if
      if y > pFloor[player] then
        pastBotEdge = 1
      end if
    end repeat
    push = "none"
    if pastRightEdge then
      pPieceLoc[player].locH = pPieceLoc[player].locH - pDiscWidth
      push = "left"
    else
      if pastLeftEdge then
        pPieceLoc[player].locH = pPieceLoc[player].locH + pDiscWidth
        push = "right"
      end if
    end if
    if pastBotEdge then
      pPieceLoc[player].locV = pPieceLoc[player].locV - pDiscHeight
      if push = "none" then
        push = "up"
        next repeat
      end if
      if push = "left" then
        push = "leftup"
        next repeat
      end if
      if push = "right" then
        push = "rightup"
      end if
    end if
  end repeat
  return push
end

on pushBackCluster me, player, pushType
  case pushType of
    "none":
    "left":
      pPieceLoc[player].locH = pPieceLoc[player].locH + pDiscWidth
    "right":
      pPieceLoc[player].locH = pPieceLoc[player].locH - pDiscWidth
    "up":
      pPieceLoc[player].locV = pPieceLoc[player].locV + pDiscHeight
    "leftup":
      pPieceLoc[player].locH = pPieceLoc[player].locH + pDiscWidth
      pPieceLoc[player].locV = pPieceLoc[player].locV + pDiscHeight
    "rightup":
      pPieceLoc[player].locH = pPieceLoc[player].locH - pDiscWidth
      pPieceLoc[player].locV = pPieceLoc[player].locV + pDiscHeight
  end case
end

on checkForOverlap me, player
  repeat with i in pActiveDiscs[player]
    repeat with j in pActiveDiscs[player]
      if i = j then
        next repeat
      end if
      if (sprite(j).locH = sprite(i).locH) and (sprite(j).locV = sprite(i).locV) then
        return 1
      end if
      if (sprite(j).locH = sprite(i).locH) and ((sprite(i).locV + pDiscHeight) > sprite(j).locV) and ((sprite(j).locV + pDiscHeight) > sprite(i).locV) then
        return 1
      end if
    end repeat
  end repeat
  return 0
end

on hitOtherPiece me, changeLoc, player
  repeat with i = 1 to pPieceSprites[player].count
    thisloc = point(changeLoc, 0) + pPieceLoc[player] + point(pPieceList[player][pPieceType[player]][pPieceOrientation[player]][i][1] * pDiscWidth, pPieceList[player][pPieceType[player]][pPieceOrientation[player]][i][2] * pDiscHeight)
    repeat with j in pActiveDiscs[player]
      if getOne(pPieceSprites[player], j) then
        next repeat
      end if
      if sprite(j).loc = thisloc then
        return 1
      end if
    end repeat
  end repeat
  return 0
end

        exit
      end if
    end if
    pPieceLoc[player].locV = pPieceLoc[player].locV + 1
    drawPiece(me, player)
  end repeat
  updateStage()
end

    if _system.milliSeconds >= (pBonusTimeStart + 1000) then
      if pBonusTime > 0 then
        pBonusTime = pBonusTime - 1
      end if
      pBonusTimeStart = _system.milliSeconds
    end if
  end if
  percent = integer(float(pBonusTime) / pBonusTimeMax * 100)
  if pBonusTime > 0 then
    sprite(37).width = pBonusMax * percent / 100
  else
    sprite(37).width = 0
  end if
  sprite(37).height = 19
  if pBonusTime > 0 then
    sprite(37).locH = 731 - (pBonusMax * percent / 100) + (pBonusMax * percent / 200)
  else
    sprite(37).locH = 731 - (pBonusMax * 100 / 100) + (pBonusMax * 100 / 200)
  end if
  member("bonusCountdown").text = "Time Left: " & pBonusTime
  if pBonusTime <= 0 then
    return 1
  else
    return 0
  end if
end

on rD2 colour
  found = 0
  repeat while found = 0
    aDisc = random(300) + pFirstSprite - 1
    case colour of
      "r":
        ranColourName = "red"
      "g":
        ranColourName = "green"
      "b":
        ranColourName = "blue"
      "y":
        ranColourName = "yellow"
      "p":
        ranColourName = "purple"
      "o":
        ranColourName = "orange"
      "c":
        ranColourName = "coal"
    end case
    inTheList = pActiveDiscs[1].getPos(aDisc) + pActiveDiscs[2].getPos(aDisc)
    if inTheList = 0 then
      sprite(aDisc).member = ranColourName
      found = 1
    end if
  end repeat
  return aDisc
end

on removeDiscs me, player
  if voidp(pRemoveList) = 1 then
    exit
  end if
  repeat with a = 1 to pRemoveList[player].count
    sprite(360 + a).locV = -100
    sprite(380 + a).locV = -100
    sprite(400 + a).locV = -100
  end repeat
  repeat with a = 1 to pRemoveList[player].count
    if sprite(pRemoveList[player][a]).member.name = "bbdisc" then
      playSound(0, "gotBBdisc.wav")
      gBBmeter[player] = gBBmeter[player] + 1
      gScore[player] = gScore[player] + 250
      showScore(me, player)
      put "me"
    end if
    deleteOne(pActiveDiscs[player], pRemoveList[player][a])
    sprite(pRemoveList[player][a]).locV = -100
  end repeat
  pRemoveList[player] = []
end

on checkIllegalDiscs me
  illegalDiscs = []
  missingDiscs = []
  overlapDiscs = []
  repeat with i = pFirstSprite to pLastSprite
    if (sprite(i).locV > 0) and (pActiveDiscs[1].getOne(i) = 0) and (pActiveDiscs[2].getOne(i) = 0) then
      illegalDiscs.add(i)
    end if
    if (sprite(i).locV < 0) and ((pActiveDiscs[1].getOne(i) <> 0) or (pActiveDiscs[2].getOne(i) <> 0)) then
      missingDiscs.add(i)
    end if
  end repeat
  repeat with i in pActiveDiscs[1]
    repeat with j in pActiveDiscs[1]
      if (sprite(i).loc = sprite(j).loc) and (i <> j) then
        put "i " & i & "j " & j
        overlapDiscs.add(i)
      end if
    end repeat
  end repeat
  if g2Play = 1 then
    repeat with i in pActiveDiscs[2]
      repeat with j in pActiveDiscs[2]
        if (sprite(i).loc = sprite(j).loc) and (i <> j) then
          overlapDiscs.add(i)
        end if
      end repeat
    end repeat
  end if
  if illegalDiscs.count > 0 then
    playSound(0, "error.wav")
    put "Illegal Discs: " & illegalDiscs
  else
  end if
  if missingDiscs.count > 0 then
    playSound(0, "error.wav")
    put "Missing Discs: " & missingDiscs
  else
  end if
  if overlapDiscs.count > 0 then
    playSound(0, "error.wav")
    put "Overlap Discs: " & overlapDiscs
  else
  end if
end

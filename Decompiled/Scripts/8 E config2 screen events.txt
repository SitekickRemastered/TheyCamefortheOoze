property pCurrentMs, pTimeset, pChange
global gPlayer, gAttackRing, gWeightDisc, gSpinGear, gBladeBase, gAttackRingName, gAttackRingCombo, gAttackRingConsec, gAttackRingChain, gWeightDiscName, gWeightDiscMass, gSpinGearName, gSpinGearStyle, gBladeBaseName, gBladeBaseStyle, g2Play, gRank, gTestMode, gARingNames, gARingCombo, gARingMulti, gARingChain, gWDiscNames, gWDiscStats, gSGearNames, gSGearStats, gBBaseNames, gBBaseStats

on idle
end

on beginSprite me
  _movie.puppetTempo(10)
  member("PressPlay").text = "Press the 'PLAY' button to start the game"
  member("PressPlay2").text = "Press the 'PLAY' button to start the game"
  lapoint = point(656, -34)
  sprite(40).loc = lapoint
  sprite(41).loc = point(735, 34)
  sprite(42).loc = lapoint
  sprite(43).loc = lapoint
  sprite(44).loc = lapoint
  pCurrentMs = _system.milliSeconds
  pChange = 0
  if g2Play = 1 then
    sprite(66).loc = point(-300, -100)
  end if
end

on exitFrame me
  if g2Play = 1 then
    dapoint = point(-18 + 4, -34)
    sprite(45).loc = dapoint
    sprite(46).loc = point(57 + 8, 34)
    sprite(47).loc = dapoint
    sprite(48).loc = dapoint
    sprite(49).loc = dapoint
    sprite(68).loc = point(137, 13)
  end if
  case gRank[1] of
    1:
      sprite(75).member = "img_player1Skill1Button_Over"
    2:
      sprite(76).member = "img_player1Skill2Button_Over"
    3:
      sprite(77).member = "img_player1Skill3Button_Over"
    4:
      sprite(78).member = "img_player1Skill4Button_Over"
    5:
      sprite(79).member = "img_player1Skill5Button_Over"
  end case
  case gRank[2] of
    1:
      sprite(80).member = "img_player2Skill1Button_Over"
    2:
      sprite(81).member = "img_player2Skill2Button_Over"
    3:
      sprite(82).member = "img_player2Skill3Button_Over"
    4:
      sprite(83).member = "img_player2Skill4Button_Over"
    5:
      sprite(84).member = "img_player2Skill5Button_Over"
  end case
  repeat with player = 1 to 2
    if player = 1 then
      fs = 25
      siteKick = 50
    else
      fs = 36
      siteKick = 55
      if g2Play = 1 then
        sprite(siteKick).loc = point(400, 300)
        sprite(siteKick + 1).loc = point(400, 300)
        sprite(siteKick + 2).loc = point(400, 300)
        sprite(siteKick + 3).loc = point(400, 300)
        sprite(siteKick + 4).loc = point(400, 300)
      end if
    end if
    case gAttackRing[player] of
      1:
        sprite(fs).member = "AttackRing1"
        sprite(siteKick + 4).member = "img_configPiece_helm3"
      2:
        sprite(fs).member = "AttackRing2"
        sprite(siteKick + 4).member = "img_configPiece_helm1"
      3:
        sprite(fs).member = "AttackRing3"
        sprite(siteKick + 4).member = "img_configPiece_helm2"
    end case
    case gWeightDisc[player] of
      1:
        sprite(fs + 1).member = "WeightDisc1"
        sprite(siteKick).member = "img_configPiece_wings3"
      2:
        sprite(fs + 1).member = "WeightDisc2"
        sprite(siteKick).member = "img_configPiece_wings2"
      3:
        sprite(fs + 1).member = "WeightDisc3"
        sprite(siteKick).member = "img_configPiece_wings1"
    end case
    case gSpinGear[player] of
      1:
        sprite(fs + 2).member = "SpinGear1"
        sprite(siteKick + 3).member = "img_configPiece_arm2"
      2:
        sprite(fs + 2).member = "SpinGear2"
        sprite(siteKick + 3).member = "img_configPiece_arm3"
      3:
        sprite(fs + 2).member = "SpinGear3"
        sprite(siteKick + 3).member = "img_configPiece_arm1"
    end case
    case gBladeBase[player] of
      1:
        sprite(fs + 3).member = "BladeBase1"
        sprite(siteKick + 2).member = "img_configPiece_legs2"
      2:
        sprite(fs + 3).member = "BladeBase2"
        sprite(siteKick + 2).member = "img_configPiece_legs1"
    end case
    wingz = sprite(siteKick).member
    skz = sprite(siteKick + 1).member
    legz = sprite(siteKick + 2).member
    armz = sprite(siteKick + 3).member
    helmz = sprite(siteKick + 4).member
    put "WINGS" & wingz
    if player = 1 then
      larect = rect(0, 0, 281, 276)
    else
      larect = rect(0, 0, 290 - 8, 276)
    end if
    if pChange <> 0 then
      member("faceWings" & player).image = member(wingz).image.crop(larect)
      member("faceSK" & player).image = member(skz).image.crop(larect)
      member("faceLegs" & player).image = member(legz).image.crop(larect)
      member("faceArms" & player).image = member(armz).image.crop(larect)
      member("faceHelm" & player).image = member(helmz).image.crop(larect)
    end if
  end repeat
  pCurrentMs = _system.milliSeconds
  if pCurrentMs > (pTimeset + 100) then
    iniLoc = the moviePath & "data\" & "sitekick.ini"
    repeat with player = 1 to 2
      gAttackRingName[player] = gARingNames[gAttackRing[player]]
      gAttackRingCombo[player] = gARingCombo[gAttackRing[player]]
      gAttackRingConsec[player] = gARingMulti[gAttackRing[player]]
      gAttackRingChain[player] = gARingChain[gAttackRing[player]]
      gWeightDiscName[player] = gWDiscNames[gWeightDisc[player]]
      gWeightDiscMass[player] = gWDiscStats[gWeightDisc[player]]
      gSpinGearName[player] = gSGearNames[gSpinGear[player]]
      gSpinGearStyle[player] = gSGearStats[gSpinGear[player]]
      gBladeBaseName[player] = gBBaseNames[gBladeBase[player]]
      gBladeBaseStyle[player] = gBBaseStats[gBladeBase[player]]
    end repeat
    pTimeset = _system.milliSeconds
  end if
  if (pChange <> 0) and (gTestMode = 1) then
    CRLF = numToChar(13) & numToChar(10)
    CR = numToChar(13)
    line1 = "Attack Ring: " & gAttackRingName[1] & " [" & gAttackRingCombo[1] & "," & gAttackRingConsec[1] & "," & gAttackRingChain[1] & "]" & CR
    line2 = "Weight Disc: " & gWeightDiscName[1] & " [" & gWeightDiscMass[1] & "]" & CR
    line3 = "Spin Gear: " & gSpinGearName[1] & " [" & gSpinGearStyle[1] & "]" & CR
    line4 = "Blade Base: " & gBladeBaseName[1] & " [" & gBladeBaseStyle[1] & "]"
    member("player1stats").text = line1 & line2 & line3 & line4
    line1 = "Attack Ring: " & gAttackRingName[2] & " [" & gAttackRingCombo[2] & "," & gAttackRingConsec[2] & "," & gAttackRingChain[2] & "]" & CR
    line2 = "Weight Disc: " & gWeightDiscName[2] & " [" & gWeightDiscMass[2] & "]" & CR
    line3 = "Spin Gear: " & gSpinGearName[2] & " [" & gSpinGearStyle[2] & "]" & CR
    line4 = "Blade Base: " & gBladeBaseName[2] & " [" & gBladeBaseStyle[2] & "]"
    member("player2stats").text = line1 & line2 & line3 & line4
    pChange = pChange - 1
    sprite(86).loc = point(400, 150)
    sprite(87).loc = point(10, 150)
  end if
  gAdminMusic(me)
  go(the frame)
end

on mouseDown
  pChange = 2
end

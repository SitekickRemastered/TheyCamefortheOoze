global gHState

on mouseEnter
  if gHState = 5 then
    set the member of sprite the currentSpriteNum to member("hiscoresButtonRematch_Over")
    sound(7).playFile(_movie.path & "sound\" & "buttonover1.wav")
  end if
end

on mouseUp
  if gHState <> 5 then
    sound(7).playFile(_movie.path & "sound\" & "warning.wav")
  end if
end

global gSpinGear, gSpinGearName, gARingNames, gARingCombo, gARingMulti, gARingChain, gWDiscNames, gWDiscStats, gSGearNames, gSGearStats, gBBaseNames, gBBaseStats

on mouseLeave
  set the member of sprite the currentSpriteNum to member("ConfigSpinGear1p")
end

on mouseUp
  sound(8).playFile(_movie.path & "sound\" & "buttonclicked2.wav")
  gSpinGear[1] = gSpinGear[1] + 1
  if gSpinGear[1] = 4 then
    gSpinGear[1] = 1
  end if
  iniLoc = the moviePath & "data\" & "sitekick.ini"
  gSpinGearName[1] = gSGearNames[gSpinGear[1]]
  member("PressPlay").text = gSpinGearName[1]
end
